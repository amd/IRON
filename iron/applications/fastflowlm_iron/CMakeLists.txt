cmake_minimum_required(VERSION 3.22)
project(iron_model_extension LANGUAGES CXX)
find_package(FastFlowLMModelExtension CONFIG REQUIRED)
find_path(BOOST_PROCESS_INCLUDE_DIR boost/process.hpp HINTS "$ENV{BOOST_ROOT}" "$ENV{CONDA_PREFIX}/Library/include" "C:/dev/boost_1_88_0" REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(iron_worker STATIC worker.cpp)
set_target_properties(iron_worker PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(iron_worker PUBLIC cxx_std_20)
target_include_directories(iron_worker PUBLIC . "${JSON_INCLUDE_DIR}" "${BOOST_PROCESS_INCLUDE_DIR}")
if(WIN32)
    target_link_libraries(iron_worker PUBLIC shell32)
endif()
add_library(iron_extension SHARED model_extension.cpp)
target_link_libraries(iron_extension PRIVATE iron_worker FastFlowLM::ModelExtension)
target_compile_definitions(iron_extension PRIVATE FLM_MODEL_EXTENSION_IMPLEMENTATION)
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:iron_extension>/model-list.json"
    INPUT "${CMAKE_CURRENT_SOURCE_DIR}/model-list.json")
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:iron_extension>/extension.json" CONTENT
    "{\"schema_version\":1,\"name\":\"iron\",\"version\":\"1\",\"library\":\"$<TARGET_FILE_NAME:iron_extension>\",\"catalog\":\"model-list.json\"}\n")
include(CTest)
set(TEST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/applications/fastflowlm_iron")
if(BUILD_TESTING AND EXISTS "${TEST_DIR}/test_worker.cpp")
    add_executable(test_iron_worker "${TEST_DIR}/test_worker.cpp")
    target_link_libraries(test_iron_worker PRIVATE iron_worker)
    add_test(NAME iron_worker COMMAND test_iron_worker)
    set_tests_properties(iron_worker PROPERTIES TIMEOUT 30 ENVIRONMENT "FLM_IRON_PYTHON=$<TARGET_FILE:test_iron_worker>")
    add_executable(test_iron_plugin "${TEST_DIR}/test_plugin.cpp")
    target_link_libraries(test_iron_plugin PRIVATE iron_extension FastFlowLM::ModelExtension)
    add_test(NAME iron_plugin COMMAND test_iron_plugin)
    set_tests_properties(iron_plugin PROPERTIES TIMEOUT 30 ENVIRONMENT
        "FLM_IRON_PYTHON=$<TARGET_FILE:test_iron_worker>;FLM_IRON_WORKER_MODULE=fake_server")
endif()
