# Optional dependencies for the stream-dse-backed operators
# (iron/operators/*_prefill_stream).
#
# Kept out of requirements.txt so an install without stream-dse still works: the
# operators' tests skip themselves (pytest.importorskip) when it is absent. CI does
# install this file (.github/actions/prereqs), so the operators run there. To build
# and run the operators and their tests:
#
#     pip install -r requirements_stream.txt
#     stream-setup-aie   # REQUIRED: installs stream-dse's pure-Python AIE codegen
#                        # dialects (xdsl-aie, snax-mlir), which cannot be PyPI
#                        # dependencies. It leaves mlir_aie/llvm-aie alone; IRON
#                        # pins those in requirements.txt.
#
# stream-dse generates each design at build time with the license-free OR-Tools
# GSCIP solver (STREAM_BACKEND selects another), and writes it under ./outputs, so
# the working directory must be writable.

onnxscript>=0.7
stream-dse>=1.15.0
